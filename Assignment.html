2.Write a function to find the largest number in an array of 20 numbers

function Problem_2(numbers){
   let largest=numbers[0];
   for(let i=0;i<20;i++){
   if(numbers[i]>largest){
   largest=numbers[i];
    }
   }
   return largest;
}
